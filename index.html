const tg = window.Telegram.WebApp; tg.ready(); tg.expand(); const user = tg.initDataUnsafe?.user; const startParam = tg.initDataUnsafe?.start_param; // вот тут приходит реферальный код if (!user) { alert('Открой через Telegram'); } // Отправляем данные на бэкенд при загрузке async function registerUser() { try { const res = await fetch('https://ТВОЙ-БЭКЕНД.com/api/register', { method: 'POST', headers: { 'Content-Type': 'application/json' }, body: JSON.stringify({ user_id: user.id, username: user.username || null, first_name: user.first_name || null, referrer: startParam || null // реферальный код }) }); const data = await res.json(); console.log('Регистрация:', data); // Можно сразу показать баланс if (data.balance !== undefined) { document.getElementById('balance').innerText = data.balance; } } catch (e) { console.error(e); } } registerUser();
